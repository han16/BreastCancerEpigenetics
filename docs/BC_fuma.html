add_min_qvalue <- function(df) {
  df$qval_min <- pmin(
    (df$la_exp.donut.qval),
    (df$la_exp.vertical.qval),
    (df$la_exp.horizontal.qval),
    (df$la_exp.lowleft.qval),
    na.rm = TRUE
  )
  return(df)
}
