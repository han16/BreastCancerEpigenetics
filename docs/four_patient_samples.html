plot_data <- data %>%
  separate(Sample_ID, into = c("Patient", "Timepoint"), sep = "_") %>%

               names_to = "Read_Pair", 
               values_to = "Counts") %>%
  mutate(
    Counts_Millions = Counts / 1e6,

      Read_Pair == "Mapped_R2" ~ "R2 Reads"
